Launcher -> LocalRegistry: create ModelRegistry

Launcher -> PerfMonitor: create PerformanceMonitor

Main -> Launcher: initialize(config)
Launcher -> ModelManager: scanAllModels()

ModelManager -> CloudRegistry: scanModels() (if configured)

== ImplementationAgent Processing ==

ImplAgent -> ImplAgent: isSimpleTask(task)?

alt Simple Task (math, short question)
    ImplAgent -> ImplAgent: handleSimpleTask(task)
    ImplAgent -> ImplAgent: createKoogAgent()
    ImplAgent -> ImplAgent: agent.run(simplePrompt) [30s timeout]
    ImplAgent --> Router: direct response
else Complex Task (requires tools)
    ImplAgent -> ImplAgent: executeWithTools(prompt, tools, maxIterations=10)
    
    loop Tool Execution Loop (max 10 iterations)
        ImplAgent -> ImplAgent: createKoogAgent() [new instance]
        ImplAgent -> ImplAgent: buildPromptWithContext(conversationHistory)
        ImplAgent -> ImplAgent: agent.run(prompt) [60s timeout per iteration]
        ImplAgent --> ImplAgent: agent response
        
        ImplAgent -> ImplAgent: extractToolCall(response)
        
        alt Tool call detected
            ImplAgent -> ImplAgent: tryParseToolCall(jsonString)
            
            alt JSON parsing successful
                ImplAgent -> ToolTracker: recordToolUsage(tool, startTime)
                ImplAgent -> KoogTools: executeTool(toolCall)
                
                alt Tool: read_file
                    KoogTools -> KoogTools: readFile(path)
                    KoogTools --> ImplAgent: file content
                else Tool: write_file
                    KoogTools -> KoogTools: writeFile(path, content)
                    KoogTools --> ImplAgent: write result
                else Tool: list_directory
                    KoogTools -> KoogTools: listDirectory(path)
                    KoogTools --> ImplAgent: directory listing
                else Tool: regex_search
                    KoogTools -> KoogTools: regexSearch(pattern, directory)
                    KoogTools --> ImplAgent: search results
                end
                
                ImplAgent -> ToolTracker: recordToolUsage(tool, success, duration)
                ImplAgent -> ImplAgent: add tool output to conversation history
                ImplAgent -> ImplAgent: continue to next iteration
            else JSON parsing failed
                ImplAgent -> ImplAgent: manual extraction fallback
                alt Manual extraction successful
                    ImplAgent -> KoogTools: executeTool(toolCall)
                    KoogTools --> ImplAgent: tool result
                    ImplAgent -> ImplAgent: continue
                else Manual extraction failed
                    ImplAgent -> ImplAgent: treat as final response
                    ImplAgent -> ImplAgent: break loop
                end
            end
        else No tool call detected
            ImplAgent -> ImplAgent: final response received
            ImplAgent -> ImplAgent: break loop
        end
        
        alt Error occurred
            alt Tool execution error
                ImplAgent -> ImplAgent: add error to conversation history
                ImplAgent -> ImplAgent: continue (allow recovery)
            else Timeout error
                alt Last iteration
                    ImplAgent -> ImplAgent: return timeout error
                else Not last iteration
                    ImplAgent -> ImplAgent: add timeout note to history
                    ImplAgent -> ImplAgent: continue
                end
            else Critical error
                ImplAgent -> ImplAgent: return error immediately
            end
        end
    end
    
    ImplAgent --> Router: AgentResponse with result
end

Router --> Orchestrator: AgentResponse
Orchestrator --> Client: response
Client --> GUI: OutputEvent flow
GUI --> User: display response

== Tool Call Extraction Strategy ==

ImplAgent -> ImplAgent: extractToolCall(response)

alt Code block found
    ImplAgent -> ImplAgent: extract JSON from ```json {...} ```
    ImplAgent -> ImplAgent: tryParseToolCall(jsonString, "code block")
else JSON object found
    ImplAgent -> ImplAgent: find JSON with "tool" field
    ImplAgent -> ImplAgent: tryParseToolCall(jsonString, "json object")
end

ImplAgent -> ImplAgent: tryParseToolCall(jsonString, source)

alt Direct JSON parsing
    ImplAgent -> ImplAgent: json.decodeFromString<ToolCall>()
    ImplAgent --> ImplAgent: tool call extracted
else Parsing failed - fix common issues
    ImplAgent -> ImplAgent: fix backslashes, paths
    ImplAgent -> ImplAgent: try parsing again
    alt Fixed parsing successful
        ImplAgent --> ImplAgent: tool call extracted
    else Still failed - manual extraction
        ImplAgent -> ImplAgent: extract tool name with regex
        ImplAgent -> ImplAgent: extract args with brace matching
        ImplAgent -> ImplAgent: create minimal JSON
        ImplAgent --> ImplAgent: tool call extracted
    end
end

== Error Handling & Fallback ==

alt ModelWrapper fails
    ModelWrapper -> Orchestrator: error
    Orchestrator -> Orchestrator: handleFallback(sessionId, error)
    Orchestrator --> Client: fallback response
else ContextProvider fails
    Context -> Orchestrator: error
    Orchestrator -> Context: retry with reduced context
    Context --> Orchestrator: minimal context
else Tool execution fails
    KoogTools -> ImplAgent: tool error
    ImplAgent -> ImplAgent: add error to conversation history
    ImplAgent -> ImplAgent: continue to next iteration (recovery)
else Agent timeout
    ImplAgent -> ImplAgent: timeout exception
    alt Simple task timeout
        ImplAgent --> Router: timeout error message
    else Complex task timeout
        alt Last iteration
            ImplAgent --> Router: timeout error message
        else Not last iteration
            ImplAgent -> ImplAgent: add timeout note, continue
        end
    end
end
